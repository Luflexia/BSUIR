Поиск женщин с ФИО на С:SELECT * FROM "Пассажир" WHERE "Пол" = 'Женский' AND "ФИО" LIKE 'С%'
Билеты дороже 100:SELECT * FROM "Билет" WHERE "Стоимость" > 100
Сотрудники по стажу:SELECT * FROM "Сотрудник" ORDER BY "Стаж" ASC
Список станций:SELECT "Название", "Город" FROM "Станция"
Маршруты длиной более 500 км:SELECT * FROM "Маршрут" WHERE "Расстояние" > 500
Сотрудники со стажем более 5 лет 10+ вагонами:SELECT "Поезд"."Компания", "Поезд"."Количество_вагонов", "Сотрудник"."ФИО", "Сотрудник"."Стаж" FROM "Поезд" CROSS JOIN "Сотрудник" WHERE "Поезд"."Количество_вагонов" > 10 AND "Сотрудник"."Стаж" > 5
Поезда с 8+ вагонами или маршруты более 300:SELECT "Поезд"."Компания", "Поезд"."Количество_вагонов", "Маршрут"."Название", "Маршрут"."Расстояние" FROM "Поезд" CROSS JOIN "Маршрут" WHERE "Поезд"."Количество_вагонов" > 8 OR "Маршрут"."Расстояние" > 300 LIMIT 30
Информация о билетах и пассажирах:SELECT "Билет".*, "Пассажир"."ФИО", "Пассажир"."Контактный_телефон" FROM "Билет" INNER JOIN "Пассажир" ON "Билет"."Пассажир_id" = "Пассажир"."id"
Расписание московских станций:SELECT "Расписание".*, "Станция"."Название" AS "Станция_отбытия" FROM "Расписание" INNER JOIN "Станция" ON "Расписание"."Станция_отбытия_id" = "Станция"."id" WHERE "Станция"."Город" = 'Москва'
Опытные сотрудники и маршруты:SELECT "Сотрудник"."ФИО", "Сотрудник"."Стаж", "Сотрудник_Маршрут"."Маршрут_id" FROM "Сотрудник" LEFT OUTER JOIN "Сотрудник_Маршрут" ON "Сотрудник"."id" = "Сотрудник_Маршрут"."Сотрудник_id" WHERE "Сотрудник"."Стаж" > 5
Скоростные поезда и сотрудники:SELECT "Поезд"."Компания", "Поезд"."Тип", "Поезд_Сотрудник"."Сотрудник_id" FROM "Поезд" LEFT OUTER JOIN "Поезд_Сотрудник" ON "Поезд"."id" = "Поезд_Сотрудник"."Поезд_id" WHERE "Поезд"."Тип" = 'Скоростной'
Станции и маршруты не в Москве:SELECT "Станция"."Название", "Станция"."Город", "Маршрут_Станция"."Маршрут_id" FROM "Маршрут_Станция" RIGHT OUTER JOIN "Станция" ON "Маршрут_Станция"."Станция_id" = "Станция"."id" WHERE "Станция"."Город" != 'Москва'
Машинисты и их поезда:SELECT "Сотрудник"."ФИО", "Сотрудник"."Должность", "Поезд_Сотрудник"."Поезд_id" FROM "Поезд_Сотрудник" RIGHT OUTER JOIN "Сотрудник" ON "Поезд_Сотрудник"."Сотрудник_id" = "Сотрудник"."id" WHERE "Сотрудник"."Должность" = 'Машинист'
Пассажиры и их билеты:SELECT "Пассажир"."ФИО", "Пассажир"."Контактный_телефон", "Билет"."Поезд_id", "Билет"."Стоимость" FROM "Пассажир" FULL OUTER JOIN "Билет" ON "Пассажир"."id" = "Билет"."Пассажир_id"
Поезда и маршруты:SELECT "Поезд"."Компания", "Поезд"."Тип", "Маршрут"."Название", "Маршрут"."Расстояние" FROM "Поезд" FULL OUTER JOIN "Маршрут" ON "Поезд"."id" = "Маршрут"."id"
Общее количество поездов:SELECT COUNT(*) AS "Общее количество поездов" FROM "Поезд"
Количество уникальных пассажиров:SELECT COUNT(DISTINCT "Пассажир_id") AS "Количество уникальных пассажиров" FROM "Билет"
Максимальная длина маршрута:SELECT MAX("Расстояние") AS "Максимальная длина маршрута (км)" FROM "Маршрут"
Максимальная стоимость билета по типу поезда:SELECT p."Тип", MAX(b."Стоимость") AS "Максимальная стоимость билета" FROM "Билет" b JOIN "Поезд" p ON b."Поезд_id" = p.id GROUP BY p."Тип"
Общая выручка от продажи билетов:SELECT SUM("Стоимость") AS "Общая выручка" FROM "Билет"
Сумма уникальных остановок:SELECT SUM(DISTINCT "Порядок_остановки") AS "Сумма уникальных остановок" FROM "Маршрут_Станция"
Количество сотрудников по должности:SELECT "Должность", COUNT(*) AS "Количество" FROM "Сотрудник" GROUP BY "Должность"
Компании с более чем 3 поездами:SELECT "Компания", COUNT(*) AS "Кол-во поездов" FROM "Поезд" GROUP BY "Компания" HAVING COUNT(*) > 3
Средний стаж по должности:SELECT "Должность", ROUND(AVG("Стаж"), 2) AS "Средний стаж" FROM "Сотрудник" GROUP BY "Должность" HAVING AVG("Стаж") > 3 ORDER BY "Средний стаж" DESC
Список пассажиров и сотрудников:SELECT "ФИО", 'Пассажир' AS "Тип" FROM "Пассажир" UNION SELECT "ФИО", 'Сотрудник' AS "Тип" FROM "Сотрудник" ORDER BY "ФИО" LIMIT 20
Москва и Минск:SELECT "Название", 'Маршрут' AS "Тип" FROM "Маршрут" WHERE "Название" LIKE '%Москва%' OR "Название" LIKE '%Минск%' UNION SELECT "Название", 'Станция' AS "Тип" FROM "Станция" WHERE "Город" = 'Москва' OR "Город" = 'Минск' ORDER BY "Название"
Сотрудники на поездах и маршрутах:SELECT "Сотрудник_id" FROM "Поезд_Сотрудник" INTERSECT SELECT "Сотрудник_id" FROM "Сотрудник_Маршрут"
Количество билетов у пассажиров:SELECT p.id, p."ФИО", COUNT(b.id) AS "Количество_билетов" FROM "Пассажир" p LEFT JOIN "Билет" b ON p.id = b."Пассажир_id" GROUP BY p.id, p."ФИО" ORDER BY "Количество_билетов" ASC
