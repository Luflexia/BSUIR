1. Показать всех сотрудников:SELECT * FROM "Сотрудник"
2. Показать все станции:SELECT * FROM "Станция"
3. Найти станции в Москве:SELECT * FROM "Станция" WHERE Город = 'Москва'
4. Найти билеты "Эконом":SELECT * FROM "Билет" WHERE Категория = 'Эконом'
5. Найти пассажиров (М):SELECT * FROM "Пассажир" WHERE Пол = 'М'
6. Найти машинистов:SELECT * FROM "Сотрудник" WHERE Должность contains 'Машинист'
7. Найти поезда РЖД:SELECT * FROM "Поезд" WHERE Компания contains 'РЖД'
8. Маршруты через СПб:SELECT * FROM "Маршрут" WHERE Название contains 'Санкт-Петербург'
9. Станции-вокзалы:SELECT * FROM "Станция" WHERE Название contains 'вокзал'
10. Поезда > 500 мест:SELECT * FROM "Поезд" WHERE Вместимость > 500
11. Билеты > 2000 руб:SELECT * FROM "Билет" WHERE Стоимость > 2000
12. Паспорта серия 45:SELECT * FROM "Пассажир" WHERE Номер_паспорта startswith '45'
13. Стаж < 5 лет:SELECT * FROM "Сотрудник" WHERE Стаж < 5
14. Путей != 4:SELECT * FROM "Станция" WHERE Количество_путей != 4
15. Билеты цена с 5:SELECT * FROM "Билет" WHERE Стоимость startswith '5'
