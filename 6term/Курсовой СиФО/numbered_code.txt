012 #define JOY_X A0       // Джойстик ось X

014 #define JOY_BTN 5      // Кнопка джойстика
015 #define BTN1 2         // Кнопка 1 (не используется)

028 #define EEPROM_BRIGHTNESS_ADDR (EEPROM_LEADERBOARD_ADDR + sizeof(leaderboard))

030 SSD1306AsciiWire display;
031 Servo servoX;
032 Servo servoY;

034 // Глобальные переменные
035 struct LeaderboardEntry {
036   char name[MAX_NAME_LEN+1];
037   unsigned long time;
038 };
039 LeaderboardEntry leaderboard[LEADERBOARD_SIZE];

041 char menuItems[MENU_ITEMS][12] = {"Start", "Records", "Settings"};
042 int menuIndex = 0;
043 int leaderboardScroll = 0;
044 int namePos = 0;
045 bool nameInputBlink = false;
046 char playerName[MAX_NAME_LEN+1] = {0};
047 char keyboard[KEYBOARD_LEN+1] = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
048 int nameCharIndex = 0;
049 unsigned long countdownStart = 0;
050 unsigned long lastJoyMove = 0;
051 unsigned long lastBlink = 0;
052 unsigned long gameStartTime = 0;
053 unsigned long gameEndTime = 0;
054 bool inCountdown = false;
055 bool gameStarted = false;
056 bool isNewRecord = false;
057 int brightness = 128;
058 bool confirmReset = false;

060 // Экраны
061 #define SCREEN_MAIN_MENU 0
062 #define SCREEN_GAME_MODE 1
063 #define SCREEN_LEADERBOARD 2

068 void setup() {
069   Serial.begin(9600);
070   delay(2000);
071   Wire.begin();
072 
073   // Настройка регистров дисплея для плавного управления яркостью
074   Wire.beginTransmission(0x3C);
075   Wire.write(0x00); // Command mode
076   Wire.write(0xD9);  // Precharge Period
077   Wire.write(0xF1);  // Default: 0xF1 (Phase 1: 1 DCLK, Phase 2: 15 DCLK)
078   Wire.write(0xDB);  // VCOMH Deselect Level
079   Wire.write(0x40);  // Default: 0x40 (~0.83*VCC)
080   Wire.endTransmission();
081 
082   display.begin(&Adafruit128x64, 0x3C);
083   display.setFont(System5x7);
084   display.clear();
085   display.setCursor(0,0);
086   display.print("TEST");
087   delay(1000);
088   display.clear();
089   pinMode(JOY_X, INPUT);
090   pinMode(JOY_Y, INPUT);
091   pinMode(JOY_BTN, INPUT_PULLUP); // D5
092   pinMode(BTN1, INPUT_PULLUP);    // D2
093   pinMode(BTN2, INPUT_PULLUP);    // D3
094   pinMode(BTN3, INPUT_PULLUP);    // D4
095   pinMode(HALL_PIN, INPUT);     // D7
096   loadLeaderboard();
097   // Загрузка яркости из EEPROM
098   EEPROM.get(EEPROM_BRIGHTNESS_ADDR, brightness);
099   // Проверка на валидность (если EEPROM пуст, будет 255 или 0)
100   if (brightness < 0 || brightness > 255) {
101       brightness = 128; // Значение по умолчанию
102   }
103   analogWrite(9, brightness); // Установить загруженную яркость
104 }
105 // EEPROM
106 void saveLeaderboard() {
107   EEPROM.put(EEPROM_LEADERBOARD_ADDR, leaderboard);
108 }
109 void loadLeaderboard() {
110   EEPROM.get(EEPROM_LEADERBOARD_ADDR, leaderboard);
111   // Проверка на пустую EEPROM
112   if (leaderboard[0].time == 0xFFFFFFFF) {
113     for (int i=0; i<LEADERBOARD_SIZE; i++) {
114       leaderboard[i].name[0] = '\0';
115       leaderboard[i].time = 0xFFFFFFFF;
116     }
117   }
118 }
119 void resetLeaderboard() {
120   for (int i=0; i<LEADERBOARD_SIZE; i++) {
121     leaderboard[i].name[0] = '\0';
122     leaderboard[i].time = 0xFFFFFFFF;
123   }
124   saveLeaderboard();
125 }
126 // Меню
127 void drawMainMenu() {
128   display.clear();
129   display.setFont(System5x7);
130   display.setCursor(0, 0);
131   display.print("LABYRINTH");
132   for (int i=0; i<MENU_ITEMS; i++) {
133     display.setCursor(0, 2 + i*2);
134     if (i == menuIndex) display.print("> "); else display.print("  ");
135     display.print(menuItems[i]);
136   }
137 }
138 void drawCountdown() {
139   display.clear();
140   display.setFont(System5x7);
141   display.setCursor(5, 3);
142   int sec = 3 - (millis() - countdownStart)/1000;
143   display.print("START: ");
144   display.print(sec);
145 }
146 void drawGameMode(unsigned long elapsed) {
147   display.setFont(fixed_bold10x15);
148   char timeStr[10];
149   dtostrf(elapsed / 1000.0, 5, 2, timeStr);
150   int textWidth = display.strWidth(timeStr);
151   int textHeight = 15;
152   int x = (SCREEN_WIDTH - textWidth) / 2;
153   int y = (SCREEN_HEIGHT / 2) - (textHeight / 2);
154   if (y < 0) y = 0;
155   int clearWidth = display.strWidth("999.99");
156   int clearX = (SCREEN_WIDTH - clearWidth) / 2;
157   if (clearX < 0) clearX = 0;
158   display.setCursor(clearX, y);
159   int spaceWidth = display.strWidth(" ");
160   int numSpaces = (spaceWidth > 0) ? (clearWidth / spaceWidth) + 1 : 10;
161   char clearStr[15]; // Ensure buffer is large enough
162   memset(clearStr, ' ', sizeof(clearStr) - 1);
163   if (numSpaces >= sizeof(clearStr)) {
164       numSpaces = sizeof(clearStr) - 1;   }
165   clearStr[numSpaces] = '\0';
166   display.print(clearStr);
167 
168   display.setCursor(x, y);
169   // display.setTextColor(WHITE);
170   display.print(timeStr);
171   display.setFont(System5x7);
172 }
173 void drawLeaderboard() {
174   display.clear();
175   display.setFont(System5x7);
176   display.setCursor(0,0);
177   display.print("LEADERBOARD");
178   for (int i=0; i<3; i++) {
179     int idx = leaderboardScroll + i;
180     if (idx >= LEADERBOARD_SIZE) break;
181     display.setCursor(0, 2+i*2);
182     display.print(idx+1);
183     display.print(". ");
184     display.print(leaderboard[idx].name);
185     display.print(" ");
186     if (leaderboard[idx].time != 0xFFFFFFFF) {
187       display.print(leaderboard[idx].time/1000.0,2);
188     } else {
189       display.print("--");
190     }
191   }
192 }
193 void drawNameInput() {
194   display.clear();
195   // Обновляем текст ввода имени с временем
196   char timeStr[20];
197   dtostrf(gameEndTime / 1000.0, 5, 2, timeStr);
198   String title = "Time: ";
199   title += timeStr;
200   title += ", enter name:";
201   int titleWidth = display.strWidth(title.c_str());
202   int titleX = (SCREEN_WIDTH - titleWidth) / 2;
203   display.setCursor(titleX, 0);
204   display.print(title.c_str());
205   // Центрируем поле ввода имени
206   int nameFieldWidth = MAX_NAME_LEN * 6;
207   int nameFieldX = (SCREEN_WIDTH - nameFieldWidth) / 2;
208   for (int i=0; i<MAX_NAME_LEN; i++) {
209     display.setCursor(nameFieldX + i*6, 2);
210     char c = playerName[i] ? playerName[i] : '_';
211     if (i == namePos && nameInputBlink && menuIndex == 0) {
212        c = ' ';
213     }
214     display.print(c);
215   }
216   // Центрируем клавиатуру (буква)
217   int keyWidth = display.strWidth("< X >");
218   int keyX = (SCREEN_WIDTH - keyWidth) / 2;
219   display.setCursor(keyX, 4);
220   if (menuIndex == 0) display.print("< "); else display.print("  ");
221   display.print(keyboard[nameCharIndex]);
222   if (menuIndex == 0) display.print(" >"); else display.print("  ");
223   // Центрируем кнопку OK
224   int okWidth = display.strWidth("> OK");
225   int okX = (SCREEN_WIDTH - okWidth) / 2;
226   display.setCursor(okX, 6);
227   if (menuIndex == 1) display.print("> "); else display.print("  ");
228   display.print("OK");
229 }
230 void drawSettings() {
231   display.clear();
232   display.setCursor(0,0);
233   display.print("SETTINGS");
234   display.setCursor(0,2);
235   if (menuIndex == 0) display.print("> "); else display.print("  ");
236   display.print("Brightness: ");
237   display.print(brightness);
238   display.setCursor(0,4);
239   if (menuIndex == 1) display.print("> "); else display.print("  ");
240   display.print("Reset Leaderboard");
241   if (confirmReset) {
242     display.setCursor(0,6);
243     display.print("Confirm? BTN2 or joyBtn");
244   }
245 }
246 // Управление
247 void loop() {
248   // Чтение входов
249   int joyX = analogRead(JOY_X);
250   int joyY = analogRead(JOY_Y);
251   bool joyBtn = digitalRead(JOY_BTN) == LOW;
252   bool btn1 = digitalRead(BTN1) == LOW; // Не используется пока
253   bool btn2 = digitalRead(BTN2) == LOW; // Кнопка Назад/ОК
254   bool btn3 = digitalRead(BTN3) == LOW; // Не используется пока
255   bool hall = digitalRead(HALL_PIN) == LOW; // Датчик Холла (финиш)
256 
257   unsigned long currentTime = millis();
258   static unsigned long lastInputTime = 0;
259   static int prevScreen = -1; // Для отслеживания смены экрана
260   bool screenNeedsUpdate = (currentScreen != prevScreen);
261   prevScreen = currentScreen;
262 
263   // Логика состояний
264   switch (currentScreen) {
265     case SCREEN_MAIN_MENU:
266       static bool joyXReleased = true;
267       if (abs(joyX - 512) < 150) joyXReleased = true;
268       if (joyX < 200 && joyXReleased) { // Влево
269         menuIndex = (menuIndex - 1 + MENU_ITEMS) % MENU_ITEMS;
270         lastInputTime = currentTime;
271         screenNeedsUpdate = true;
272         joyXReleased = false;
273       } else if (joyX > 800 && joyXReleased) { // Вправо
274         menuIndex = (menuIndex + 1) % MENU_ITEMS;
275         lastInputTime = currentTime;
276         screenNeedsUpdate = true;
277         joyXReleased = false;
278       }
279       if (joyBtn || btn2) { // Выбор
280         if (menuIndex == 0) { // Start
281           currentScreen = SCREEN_GAME_MODE;
282           inCountdown = true;
283           countdownStart = currentTime;
284           gameStarted = false;
285           // servoX.attach(SERVO_X_PIN); // Подключаем сервы ПОСЛЕ отсчета
286           // servoY.attach(SERVO_Y_PIN);
287         } else if (menuIndex == 1) { // Records
288           currentScreen = SCREEN_LEADERBOARD;
289           leaderboardScroll = 0;
290         } else if (menuIndex == 2) { // Settings
291           currentScreen = SCREEN_SETTINGS;
292           menuIndex = 0; // Сброс индекса для меню настроек
293           confirmReset = false;
294         }
295         screenNeedsUpdate = true;
296         delay(300); // Антидребезг кнопки
297       }
298       if (screenNeedsUpdate) drawMainMenu();
299       break;
300 
301     case SCREEN_GAME_MODE:
302       if (inCountdown) {
303         if (currentTime - countdownStart >= 3000) {
304           inCountdown = false;
305           gameStarted = true;
306           gameStartTime = currentTime;
307           servoX.attach(SERVO_X_PIN); // Подключаем сервы здесь
308           servoY.attach(SERVO_Y_PIN);
309           servoX.write(90); // Устанавливаем начальное положение X
310           servoY.write(90); // Устанавливаем начальное положение Y
311           display.clear(); // Очищаем экран от надписи "START"
312           screenNeedsUpdate = true;
313         } else {
314           // Обновляем экран каждую секунду во время отсчета
315           static unsigned long lastCountdownDraw = 0;
316           if (currentTime - lastCountdownDraw > 1000) {
317              drawCountdown();
318              lastCountdownDraw = currentTime;
319           }
320         }
321       } else if (gameStarted) {
322         unsigned long elapsedTime = currentTime - gameStartTime;
323         // Y управляет D10 (SERVO_X_PIN), X управляет D8 (SERVO_Y_PIN)
324         const int DEAD_ZONE_LOW = 400;
325         const int DEAD_ZONE_HIGH = 600;
326         int servoAngleX = 90; // Нейтральное положение
327         int servoAngleY = 90; // Нейтральное положение
328 
329         // Управление Servo X (ось Y джойстика -> SERVO_X_PIN/D10)
330 static float smoothServoX = 90;
331 static float smoothServoY = 90;
332 const float SMOOTH_STEP = 0.1; // более плавного отклика
333 if (joyY < DEAD_ZONE_LOW) {
334   servoAngleX = map(joyY, 0, DEAD_ZONE_LOW - 1, 105, 90);
335 } else if (joyY > DEAD_ZONE_HIGH) {
336   servoAngleX = map(joyY, DEAD_ZONE_HIGH + 1, 1023, 90, 75);
337 }
338 // Инвертированное управление Servo Y (ось X джойстика -> SERVO_Y_PIN/D8)
339 if (joyX < DEAD_ZONE_LOW) {
340   servoAngleY = map(joyX, 0, DEAD_ZONE_LOW - 1, 75, 90);
341 } else if (joyX > DEAD_ZONE_HIGH) {
342   servoAngleY = map(joyX, DEAD_ZONE_HIGH + 1, 1023, 90, 105);
343 }
344 // Плавное движение серв
345 // Плавное изменение угла с учетом предыдущего значения
346 smoothServoX = smoothServoX + (servoAngleX - smoothServoX) * SMOOTH_STEP;
347 smoothServoY = smoothServoY + (servoAngleY - smoothServoY) * SMOOTH_STEP;
348 servoX.write(constrain((int)(smoothServoX + 0.5), 60, 120));
349 servoY.write(constrain((int)(smoothServoY + 0.5), 60, 120));
350         if (hall) { // Финиш
351           gameEndTime = elapsedTime;
352           gameStarted = false;
353           servoX.detach(); // Отключаем сервы после игры
354           servoY.detach();
355           // Проверка на новый рекорд
356           isNewRecord = false;
357           for(int i=0; i<LEADERBOARD_SIZE; i++) {
358             if (gameEndTime < leaderboard[i].time) {
359               isNewRecord = true;
360               break;
361             }
362           }
363           if (isNewRecord) {
364             currentScreen = SCREEN_NAME_INPUT;
365             memset(playerName, 0, sizeof(playerName));
366             namePos = 0;
367             nameCharIndex = 0;
368           } else {
369             currentScreen = SCREEN_MAIN_MENU; // Возврат в меню, если не рекорд
370           }
371           screenNeedsUpdate = true;
372         }
373         // Обновляем время на экране раз в 100мс
374         static unsigned long lastGameDraw = 0;
375         if (currentTime - lastGameDraw > 100) {
376            drawGameMode(elapsedTime);
377            lastGameDraw = currentTime;
378         }
379 
380       } else { // Если игра закончилась (не через датчик Холла, например, по кнопке)
381          // Пока не реализовано
382       }
383       if (btn2) { // Выход из игры (добавить подтверждение?)
384           gameStarted = false;
385           inCountdown = false;
386           servoX.detach();
387           servoY.detach();
388           currentScreen = SCREEN_MAIN_MENU;
389           screenNeedsUpdate = true;
390           delay(300);
391       }
392       break;
393 
394     case SCREEN_LEADERBOARD:
395        if (currentTime - lastInputTime > 200) { // Задержка для джойстика
396          if (joyX < 200 && leaderboardScroll > 0) { // Влево
397            leaderboardScroll--;
398            lastInputTime = currentTime;
399            screenNeedsUpdate = true;
400          } else if (joyX > 800 && leaderboardScroll < LEADERBOARD_SIZE - 3) { // Вправо (теперь X)
401            leaderboardScroll++;
402            lastInputTime = currentTime;
403            screenNeedsUpdate = true;
404          }
405        }
406        if (btn2 || joyBtn) { // Назад
407          currentScreen = SCREEN_MAIN_MENU;
408          screenNeedsUpdate = true;
409          delay(300);
410        }
411        if (screenNeedsUpdate) drawLeaderboard();
412        break;
413 
414     case SCREEN_NAME_INPUT:
415       // Логика ввода имени
416       static bool nameInputJoyYReleased = true;
417       static bool nameInputJoyXReleased = true;
418       static unsigned long lastRepeat = 0;
419       if (currentTime - lastInputTime > 60) {
420         // Переключение между полем ввода и кнопкой OK по X
421         if (abs(joyX - 512) < 150) nameInputJoyXReleased = true;
422         if (joyX < 200 && nameInputJoyXReleased) { // Влево
423           menuIndex = 0;
424           nameInputJoyXReleased = false;
425           lastInputTime = currentTime;
426           screenNeedsUpdate = true;
427         } else if (joyX > 800 && nameInputJoyXReleased) { // Вправо
428           menuIndex = 1;
429           nameInputJoyXReleased = false;
430           lastInputTime = currentTime;
431           screenNeedsUpdate = true;
432         }
433         // Прокрутка букв по Y (теперь Y)
434         if (menuIndex == 0) {
435           if (joyY < 200) { // Вверх
436             if (currentTime - lastRepeat > 80) {
437               nameCharIndex = (nameCharIndex - 1 + KEYBOARD_LEN) % KEYBOARD_LEN;
438               lastRepeat = currentTime;
439               screenNeedsUpdate = true;
440             }
441           } else if (joyY > 800) { // Вниз
442             if (currentTime - lastRepeat > 80) {
443               nameCharIndex = (nameCharIndex + 1) % KEYBOARD_LEN;
444               lastRepeat = currentTime;
445               screenNeedsUpdate = true;
446             }
447           } else {
448             lastRepeat = 0;
449           }
450         }
451       }
452       // Мигание курсора
453       if (currentTime - lastBlink > 400) {
454         nameInputBlink = !nameInputBlink;
455         lastBlink = currentTime;
456         screenNeedsUpdate = true;
457       }
458       // Подтверждение выбора
459       if ((joyBtn || btn2) && menuIndex == 0 && namePos < MAX_NAME_LEN) {
460         playerName[namePos] = keyboard[nameCharIndex];
461         namePos++;
462         screenNeedsUpdate = true;
463         delay(120);
464       } else if ((joyBtn || btn2) && menuIndex == 1) {
465         if (namePos > 0) {
466           int insertPos = -1;
467           for (int i = 0; i < LEADERBOARD_SIZE; i++) {
468               if (gameEndTime < leaderboard[i].time) {
469                   insertPos = i;
470                   break;
471               }
472           }
473           if (insertPos != -1) {
474               for (int i = LEADERBOARD_SIZE - 1; i > insertPos; i--) {
475                   leaderboard[i] = leaderboard[i - 1];
476               }
477               strncpy(leaderboard[insertPos].name, playerName, MAX_NAME_LEN);
478               leaderboard[insertPos].name[MAX_NAME_LEN] = '\0';
479               leaderboard[insertPos].time = gameEndTime;
480               saveLeaderboard();
481           }
482         }
483         currentScreen = SCREEN_LEADERBOARD;
484         leaderboardScroll = 0;
485         screenNeedsUpdate = true;
486         delay(300);
487       }
488       if (screenNeedsUpdate) drawNameInput();
489       break;
490 
491     case SCREEN_SETTINGS:
492       if (currentTime - lastInputTime > 200) {
493         if (joyX < 200) { // Влево (теперь по X)
494           menuIndex = (menuIndex - 1 + 2) % 2; // 2 пункта в настройках
495           lastInputTime = currentTime;
496           confirmReset = false; // Сброс подтверждения при смене пункта
497           screenNeedsUpdate = true;
498         } else if (joyX > 800) { // Вправо
499           menuIndex = (menuIndex + 1) % 2;
500           lastInputTime = currentTime;
501           confirmReset = false;
502           screenNeedsUpdate = true;
503         }
504 
505         if (menuIndex == 0) { // Яркость
506           if (joyY < 200 && brightness > 0) { // Вверх
507             brightness -= 8;
508             if (brightness < 0) brightness = 0;
509             analogWrite(9, brightness);
510 
511             Wire.beginTransmission(0x3C);
512             Wire.write(0x00); // Command mode
513             Wire.write(0x81); // Set Contrast Control
514             Wire.write(brightness);
515             Wire.write(0xD9);  // Precharge Period
516             Wire.write(0xF1 - (brightness/4));
517             Wire.endTransmission();
518 
519             lastInputTime = currentTime;
520             screenNeedsUpdate = true;
521           } else if (joyY > 800 && brightness < 255) { // Вниз
522             brightness += 8;
523             if (brightness > 255) brightness = 255;
524             analogWrite(9, brightness);
525 
526             Wire.beginTransmission(0x3C);
527             Wire.write(0x00); // Command mode
528             Wire.write(0x81); // Set Contrast Control
529             Wire.write(brightness);
530             Wire.write(0xD9);  // Precharge Period
531             Wire.write(0xF1 - (brightness/4));
532             Wire.endTransmission();
533 
534             lastInputTime = currentTime;
535             screenNeedsUpdate = true;
536           }
537         } else if (menuIndex == 1) { // Сброс рекордов
538           if ((joyBtn || btn2) && !confirmReset) {
539             confirmReset = true;
540             lastInputTime = currentTime;
541             screenNeedsUpdate = true;
542             delay(300);
543           }
544         }
545       }
546 
547       // Обработка действий после навигации/изменений
548       if (confirmReset) {
549           if (joyBtn || btn2) {
550               resetLeaderboard();
551               confirmReset = false;
552               screenNeedsUpdate = true;
553               delay(300);
554           }
555       } else {
556           // Выход по BTN2 или кнопке джойстика
557           if (btn2 || (menuIndex == 0 && joyBtn)) {
558               EEPROM.put(EEPROM_BRIGHTNESS_ADDR, brightness);
559               currentScreen = SCREEN_MAIN_MENU;
560               screenNeedsUpdate = true;
561               delay(300);
562           }
563       }
564       if (screenNeedsUpdate) drawSettings();
565       break;
566   }
567 
568   delay(10);
569 }
